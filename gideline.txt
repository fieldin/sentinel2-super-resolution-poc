redeploy local docker (from repo root):
docker rm -f up42-poc >/dev/null 2>&1 || true && docker build -t up42-poc:local -f Dockerfile . && docker run -d -p 8080:8080 --name up42-poc --env-file .env -v "$(pwd)/data:/app/data" up42-poc:local


deploy to kubernetes (from repo root):
./deploy.sh latest 2>&1

start the fetch:
# Start the services
make up

# Check it's running
curl http://localhost:8080/health

# Run the pipeline
make pipeline
